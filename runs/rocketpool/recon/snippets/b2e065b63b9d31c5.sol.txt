            return UpgradeProposalState.Vetoed;
        } else if (getExecuted(_upgradeProposalID)) {
            return UpgradeProposalState.Executed;
        } else if (block.timestamp < getEnd(_upgradeProposalID)) {
            return UpgradeProposalState.Pending;
