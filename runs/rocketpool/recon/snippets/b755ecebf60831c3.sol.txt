function efficientSha256(bytes32 _left, bytes32 _right) internal view returns (bytes32 ret) {
        assembly {
            mstore(0x00, _left)
            mstore(0x20, _right)

            let result := staticcall(gas(), 0x02, 0x00, 0x40, 0x00, 0x20)
            if iszero(result) {
                revert(0,0)
            }

            ret := mload(0x00)
        }
    }
